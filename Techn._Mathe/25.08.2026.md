## 1. Berechnen Sie die molaren Massen unter Berücksichtigung der Rundungsregeln

a. NiCl2
Ni = 58.693
Cl = 35.45

M(NiCl2) = 58.693 + 2*(35.45)
= 129.593 g/mol

b. Al2O3
Al = 26.982
O = 15.999

M(Al2O3) = 2*(26.982) + 3*(15.999)
= 101.961 g/mol

c. Ca3(PO4)2
Ca = 40.078
P = 30.974
O = 15.999

M(Ca3(PO4)2) = 3*(40.078) + 2*(30.974+(4*15.999))
= 310,174 g/mol

d. NO3
N = 14.007
O = 15.999

M(NO3) = 14.007 + 3*15.999
= 62.004 g/mol

m = M* n

n = 35g / 26.982g/mol
